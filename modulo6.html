Revisão básica

1) Leia dois números inteiros e mostre a soma, subtração, multiplicação e divisão.


#include <stdio.h>

int main() {
  int a;
  int b;

  printf("Digite o primeiro numero: ");
  scanf("%d", &a);

  printf("Digite o segundo numero: ");
  scanf("%d", &b);

  printf("Soma: %d\n", a + b);
  printf("Subtracao: %d\n", a - b);
  printf("Multiplicacao: %d\n", a * b);

  if (b != 0) {
    printf("Divisao: %.2f\n", (float)a / b);
  } else {
    printf("Nao existe divisao por zero.\n");
  }

  return 0;
}

2) Leia uma idade e informe se a pessoa é maior de idade.


#include <stdio.h>

int main() {
  int idade;

  printf("Digite a idade: ");
  scanf("%d", &idade);

  if (idade >= 18) {
    printf("Maior de idade");
  } else {
    printf("Menor de idade");
  }

  return 0;
}

Laços e vetores

3) Mostre a tabuada de um número informado pelo usuário.


#include <stdio.h>

int main() {
  int numero;
  int i;

  printf("Digite um numero: ");
  scanf("%d", &numero);

  for (i = 1; i <= 10; i++) {
    printf("%d x %d = %d\n", numero, i, numero * i);
  }

  return 0;
}

4) Leia 5 notas em um vetor e calcule a média.


#include <stdio.h>

int main() {
  float notas[5];
  float soma = 0;
  int i;

  for (i = 0; i < 5; i++) {
    printf("Digite a nota %d: ", i + 1);
    scanf("%f", &notas[i]);
    soma = soma + notas[i];
  }

  printf("Media: %.2f", soma / 5);

  return 0;
}

Funções e ponteiros

5) Crie uma função que receba um número e retorne o dobro.


#include <stdio.h>

int dobrar(int numero) {
  return numero * 2;
}

int main() {
  int valor;

  printf("Digite um numero: ");
  scanf("%d", &valor);

  printf("Dobro: %d", dobrar(valor));

  return 0;
}

6) Use ponteiro para alterar o valor de uma variável dentro de uma função.


#include <stdio.h>

void alterarValor(int *numero) {
  *numero = 100;
}

int main() {
  int valor = 10;

  printf("Antes: %d\n", valor);

  alterarValor(&valor);

  printf("Depois: %d\n", valor);

  return 0;
}
